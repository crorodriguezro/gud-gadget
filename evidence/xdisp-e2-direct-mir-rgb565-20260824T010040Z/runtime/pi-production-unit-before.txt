# /etc/systemd/system/gud-userspace.service
[Unit]
Description=Userspace GUD gadget
After=systemd-modules-load.service local-fs.target
Wants=systemd-modules-load.service
ConditionPathExists=/home/cristian/gud-drm
ConditionPathExists=/dev/dri/card0

[Service]
Type=simple
User=root
WorkingDirectory=/home/cristian
Environment=RUST_LOG=info
ExecStartPre=/bin/sh -c 'while [ ! -e /dev/dri/card0 ] || [ ! -e /sys/class/udc/3f980000.usb/state ]; do sleep 1; done'
ExecStart=/home/cristian/gud-drm /dev/dri/card0
Restart=on-failure
RestartSec=1s
StartLimitIntervalSec=0

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/gud-userspace.service.d/10-xdisp-p0.1-containment.conf
# XDISP-P0.1 containment: a proven-idle detach stays in the persistent
# FunctionFS process. Non-zero failures and crashes stay down.
[Service]
Restart=no
# An in-flight or poisoned Step 5 process ignores SIGTERM so userspace does not
# trigger the known-risk teardown path. Do not let systemd escalate that stop
# to SIGKILL; recover by physical/hardware reset instead.
SendSIGKILL=no

# /etc/systemd/system/gud-userspace.service.d/20-xdisp-p0.1-ffs-read-size.conf
# XDISP-P0.1 Step 5: replace 125 512-byte reads with four larger reads for the
# normal 64,000-byte tile while limiting contiguous allocation pressure.
[Service]
Environment=GUD_FFS_READ_SIZE=16384
Environment=RUST_LOG=debug
